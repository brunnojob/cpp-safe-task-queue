cmake_minimum_required(VERSION 3.20)
project(safe_task_queue LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(task_queue src/main.cpp)
target_include_directories(task_queue PRIVATE include)
target_link_libraries(task_queue PRIVATE Threads::Threads)
target_compile_options(task_queue PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_executable(queue_tests tests/queue_test.cpp)
target_include_directories(queue_tests PRIVATE include)
target_link_libraries(queue_tests PRIVATE Threads::Threads)
add_test(NAME queue COMMAND queue_tests)
